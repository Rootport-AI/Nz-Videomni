+def h3_stage_window(events: list, name: str) -> dict:
+    """One H3 stage's span from server.log: [the "started" line -> the last line].
+
+    The H3 worker emits each stage's index-0 event BEFORE any work (encode
+    index 0, ``stage1_denoise`` 0/N before step 1, decode 0), so the started
+    line opens the window and the final line closes it: ``text encode`` ends
+    on ``5/4`` with keyframes (any K; engine_h3 encode_steps) / ``4/3`` for T2V,
+    ``stage-1 denoise`` on ``N/N``, ``VAE decode`` on ``4/3 steps`` (after the
+    mp4 write).  Lines in between are thinned to one per 2 s, but the first
+    and the last always appear (ltx/adapter.py _log_stage_progress).
+    Because the denoise started line comes before step 1, the per-step time
+    is ``elapsed / last_done`` -- NOT the LTX ``elapsed / (last_done - 1)``
+    of :func:`stage_groups`'s own ``sec_per_step``.
+    """
+    groups = stage_groups(events, H3_STAGE_LABELS[name])
+    if not groups:
+        return {"name": name, "label": H3_STAGE_LABELS[name], "found": False,
+                "start": None, "end": None}
+    grp = groups[-1]
+    done, total = grp.get("last_done"), grp.get("total")
+    elapsed = grp.get("log_elapsed")
+    out = {"name": name, "label": H3_STAGE_LABELS[name], "found": True,
+           "groups_seen": len(groups),
+           "start": grp["start_t"], "end": grp["end_t"],
+           "start_iso": grp["start_iso"], "end_iso": grp["end_iso"],
+           "lines": grp["lines"], "last_done": done, "total": total,
+           "log_elapsed_s": elapsed,
+           "completed": bool(done is not None and total and done >= total)}
+    if name == "denoise":
+        out["sec_per_step_log"] = (rnd(elapsed / done, 3)
+                                   if (elapsed is not None and done) else None)
+    return out
